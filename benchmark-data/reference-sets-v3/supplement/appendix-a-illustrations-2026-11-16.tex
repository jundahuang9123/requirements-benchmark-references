  font=\scriptsize,
  box/.style={draw, rounded corners=2pt, align=center, minimum height=12mm, text width=21mm, inner sep=1.2pt, fill=white, execute at begin node={\hyphenpenalty=10000}},
  med/.style={box, fill=gray!12},
  prog/.style={box, fill=gray!28, minimum height=10.5mm},

\node[box] (case) {\textbf{Case}: up to six needs with one task label};
\node[box, right=2mm of case] (r1) {\textbf{Round 1}, roles alone: concerns, at most one experience claim};
\node[med, right=2mm of r1] (draft) {\textbf{Mediator draft}: an action per need, a decision per claim};
\node[box, dashed, right=2mm of draft] (r2) {\textbf{Round 2} if needed: roles reply};
\node[med, dashed, right=2mm of r2] (final) {\textbf{Mediator}: final plan};
\node[prog, below=8mm of final] (check) {\textbf{Separate check} of every action};
\node[prog, left=2mm of check] (gates) {\textbf{Program gates}: sources, base labels, conflicts};
\node[outc, left=6mm of gates] (rew) {Rewritten, then rechecked};
\node[outc, above=1.1mm of rew] (kept) {Kept or set aside};
\node[outc, below=1.1mm of rew] (person) {To a person};
\draw[arr] (case) -- (r1); \draw[arr] (r1) -- (draft); \draw[arr] (draft) -- (r2); \draw[arr] (r2) -- (final);
\draw[arr] (final) -- (check); \draw[arr] (check) -- (gates);
\coordinate (bus) at ($(gates.west)+(-3mm,0)$);
\draw[semithick] (gates.west) -- (bus);
\draw[arr] (bus) |- (kept.east);
\draw[arr] (bus) -- (rew.east);
\draw[arr] (bus) |- (person.east);
\end{tikzpicture}
\caption{One discussion case, formed by the task labels. Concerns may cite the case's passages and, from round 2, passages retrieved for the roles' queries; an experience claim cites none. Round 2 and the final plan run only when the draft changes or proposes a need or meets an objection. Needs proposed from experience alone go to a person unapplied. Light: mediator; dark: checker and program rules.}
\label{fig:discussion}
